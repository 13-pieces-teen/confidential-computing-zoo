//go:build argus_experiment_static

// Compiled only through the experiment Go overlay, never the normal publisher.
package clientcredentials

import (
	"context"
	"encoding/json"
	"encoding/pem"
	"errors"
	"fmt"
	"os"
	"strings"
	"time"

	api "github.com/spiffe/go-spiffe/v2/exp/proto/spiffe/broker"
	"github.com/spiffe/go-spiffe/v2/spiffeid"
	"github.com/spiffe/spiffe-helper/pkg/broker"
)

func experimentDER(data []byte, kind string) ([]byte, error) {
	var result []byte
	for len(strings.TrimSpace(string(data))) != 0 {
		block, rest := pem.Decode(data)
		if block == nil || block.Type != kind {
			return nil, fmt.Errorf("invalid static PEM material")
		}
		result = append(result, block.Bytes...)
		data = rest
	}
	if len(result) == 0 {
		return nil, fmt.Errorf("empty static PEM material")
	}
	return result, nil
}

func experimentMaterial(cert, key, bundle []byte, target string, now time.Time) (*broker.Credentials, error) {
	id, err := spiffeid.FromString(target)
	if err != nil || !strings.Contains(id.Path(), "/experiment/") || !strings.HasSuffix(id.Path(), "/static") {
		return nil, fmt.Errorf("static client must use its isolated experiment identity")
	}
	certDER, err := experimentDER(cert, "CERTIFICATE")
	if err != nil {
		return nil, err
	}
	keyDER, err := experimentDER(key, "PRIVATE KEY")
	if err != nil {
		return nil, err
	}
	bundleDER, err := experimentDER(bundle, "CERTIFICATE")
	if err != nil {
		return nil, err
	}
	return broker.Snapshot(&api.SubscribeToX509SVIDResponse{Svids: []*api.X509SVID{{
		SpiffeId: target, X509Svid: certDER, X509SvidKey: keyDER, Bundle: bundleDER,
	}}}, id, now)
}

func RunExperimentStatic(ctx context.Context, c Config, certPath, keyPath, bundlePath string) (result error) {
	if err := c.validate(); err != nil {
		return err
	}
	paths := []string{certPath, keyPath, bundlePath}
	var contents [][]byte
	for _, path := range paths {
		if err := protected(path, false); err != nil {
			return err
		}
		info, err := os.Stat(path)
		if err != nil {
			return err
		}
		if info.Size() > 65536 {
			return fmt.Errorf("static material exceeds size bound")
		}
		if path == keyPath && info.Mode().Perm()&0077 != 0 {
			return fmt.Errorf("static private key must be root-only")
		}
		data, err := os.ReadFile(path)
		if err != nil {
			return err
		}
		contents = append(contents, data)
	}
	material, err := experimentMaterial(contents[0], contents[1], contents[2], c.TargetID, time.Now())
	if err != nil {
		return err
	}
	if err := os.MkdirAll(c.Directory, 0750); err != nil {
		return err
	}
	if err := checkFilesystem(c.Directory); err != nil {
		return err
	}
	p := &publisher{dir: c.Directory, gid: c.ReaderGID}
	if err := p.access(p.dir, 0750); err != nil {
		return err
	}
	if err := p.clear(); err != nil {
		return err
	}
	defer func() { result = errors.Join(result, p.clear()) }()
	if err := protected(c.Registration, false); err != nil {
		return err
	}
	bytes, err := os.ReadFile(c.Registration)
	if err != nil {
		return err
	}
	var target Target
	if err := json.Unmarshal(bytes, &target); err != nil {
		return err
	}
	check := func() error {
		current, err := observe(target.PID)
		if err != nil || current != target {
			return fmt.Errorf("registered static-client instance changed")
		}
		if !time.Now().Before(material.Expires) {
			return fmt.Errorf("static client certificate expired")
		}
		return nil
	}
	if err := check(); err != nil {
		return err
	}
	if err := p.publish(material, time.Now()); err != nil {
		return err
	}
	ticker := time.NewTicker(250 * time.Millisecond)
	defer ticker.Stop()
	for {
		select {
		case <-ctx.Done():
			return context.Cause(ctx)
		case <-ticker.C:
			// Completed target observation is required before extending the
			// same 2.5s client lease. No Broker or Workload API is contacted.
			if err := check(); err != nil {
				return err
			}
			if err := p.renew(time.Now()); err != nil {
				return err
			}
		}
	}
}
