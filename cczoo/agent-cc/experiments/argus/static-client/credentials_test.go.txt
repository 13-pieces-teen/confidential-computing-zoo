//go:build argus_experiment_static

package clientcredentials

import (
	"encoding/json"
	"encoding/pem"
	"os"
	"path/filepath"
	"testing"
	"time"

	"github.com/spiffe/go-spiffe/v2/spiffeid"
)

func TestExperimentStaticMaterialAndLease(t *testing.T) {
	saved := clientID
	clientID = spiffeid.RequireFromString("spiffe://argus.local/client/a/experiment/trial/static")
	defer func() { clientID = saved }()
	source := snapshot(t).Svids[0]
	certificate := pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: source.X509Svid})
	key := pem.EncodeToMemory(&pem.Block{Type: "PRIVATE KEY", Bytes: source.X509SvidKey})
	bundle := pem.EncodeToMemory(&pem.Block{Type: "CERTIFICATE", Bytes: source.Bundle})
	material, err := experimentMaterial(certificate, key, bundle, clientID.String(), time.Now())
	if err != nil {
		t.Fatal(err)
	}
	p := &publisher{dir: t.TempDir(), gid: -1}
	if err := p.publish(material, time.Now()); err != nil {
		t.Fatal(err)
	}
	data, err := os.ReadFile(filepath.Join(p.dir, "ready.json"))
	if err != nil {
		t.Fatal(err)
	}
	var current lease
	if err := json.Unmarshal(data, &current); err != nil {
		t.Fatal(err)
	}
	if current.Until > time.Now().Add(2500*time.Millisecond).UnixMilli() || current.Serial != "1f" {
		t.Fatal("wrong static lease profile")
	}
	if err := p.renew(material.Expires); err == nil {
		t.Fatal("expired static credentials renewed")
	}
	if err := p.clear(); err != nil {
		t.Fatal(err)
	}
	for _, id := range []string{"spiffe://argus.local/client/production", "spiffe://argus.local/client/b/experiment/trial/static"} {
		if _, err := experimentMaterial(certificate, key, bundle, id, time.Now()); err == nil {
			t.Fatal("wrong/production ID allowed")
		}
	}
	second := snapshot(t).Svids[0]
	wrong := pem.EncodeToMemory(&pem.Block{Type: "PRIVATE KEY", Bytes: second.X509SvidKey})
	if _, err := experimentMaterial(certificate, wrong, bundle, clientID.String(), time.Now()); err == nil {
		t.Fatal("wrong key allowed")
	}
	if _, err := experimentMaterial(certificate, key, bundle, clientID.String(), time.Now().Add(2*time.Hour)); err == nil {
		t.Fatal("expired certificate allowed")
	}
}
